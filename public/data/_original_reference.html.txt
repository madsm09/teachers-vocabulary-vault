function loadContent() {
    const subject = document.getElementById('subjectSelect').value;
    const display = document.getElementById('displayCard');
    
    if (!subject) {
        alert("Please select a subject first!");
        return;
    }
    
    const info = data[subject];
    document.getElementById('subjectTitle').innerText = info.title;
    document.getElementById('gameRecommendation').innerText = info.game;
    
    let html = '';
    info.riddles.forEach((item, index) => {
        html += `
            <div class="riddle-item">
                <h4>Case ${index < 9 ? '0' + (index + 1) : (index + 1)} // Target: ${item.term}</h4>
                <p><strong>Riddle:</strong> "${item.riddle}"</p>
                <div class="hint">Coded Hint: ${item.hint}</div>
            </div>
        `;
    });
    
    document.getElementById('riddlesContainer').innerHTML = html;
    display.style.display = 'block';
}
